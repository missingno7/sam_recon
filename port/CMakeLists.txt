cmake_minimum_required(VERSION 3.20)
project(sam_milestone2 C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(SAM_OPTIMIZATION "0" CACHE STRING "GCC optimization level, 0 or 2")
set_property(CACHE SAM_OPTIMIZATION PROPERTY STRINGS 0 2)
if(NOT SAM_OPTIMIZATION MATCHES "^[02]$")
  message(FATAL_ERROR "SAM_OPTIMIZATION must be 0 or 2")
endif()
get_filename_component(SNAPSHOT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(REPO_ROOT "${SNAPSHOT_ROOT}")
while(NOT EXISTS "${REPO_ROOT}/game/SAM.PAS")
  get_filename_component(REPO_PARENT "${REPO_ROOT}/.." ABSOLUTE)
  if(REPO_PARENT STREQUAL REPO_ROOT)
    message(FATAL_ERROR "Cannot find canonical game/SAM.PAS above ${SNAPSHOT_ROOT}")
  endif()
  set(REPO_ROOT "${REPO_PARENT}")
endwhile()
set(SDL3_ROOT "C:/tools/sdl3-3.4.16-mingw/SDL3-3.4.16/x86_64-w64-mingw32" CACHE PATH "SDL3 installation")
if(WIN32)
  set(SDL3_DIR "${SDL3_ROOT}/lib/cmake/SDL3" CACHE PATH "SDL3 package")
endif()
find_package(SDL3 CONFIG REQUIRED)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
if(NOT MSVC)
  add_compile_options(-Wall -Wextra -Wpedantic "-O${SAM_OPTIMIZATION}" -fno-strict-aliasing)
endif()
add_library(sam_platform STATIC
  pc/pc.c pc/pc_io.c pc/pc_dos.c pc/pc_sdl.c pc/pc_speaker.c pc/vga.c pc/fonts.c
  tp/crt.c tp/dos.c tp/system.c)
target_include_directories(sam_platform PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/pc" "${CMAKE_CURRENT_SOURCE_DIR}/tp")
target_link_libraries(sam_platform PUBLIC SDL3::SDL3)
if(NOT MSVC)
  target_link_libraries(sam_platform PUBLIC m)
endif()
file(GLOB LIB_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/lib/*.c")
add_library(sam_lib STATIC ${LIB_SOURCES})
target_link_libraries(sam_lib PUBLIC sam_platform)
file(GLOB PASCAL_SOURCE CONFIGURE_DEPENDS "${REPO_ROOT}/game/*.PAS")
file(GLOB TRANSLATOR_SOURCE CONFIGURE_DEPENDS "${SNAPSHOT_ROOT}/tools/pas2c*.py")
set(GENERATED)
set(SAM_PLACEMENT "${SNAPSHOT_ROOT}/config/port.json" CACHE FILEPATH "Committed episode placement; no DOS toolchain needed")
foreach(episode sam1 sam2 sam3)
  set(GEN "${CMAKE_BINARY_DIR}/port-gen/${episode}")
  set(EPISODE_SOURCES)
  foreach(module SAM PROGRPHX SAMDATA SAMSOUND GAMEJOY PCXGENUS TEXT80 SAMUI SAMLEVEL SAMPLAY loader)
    list(APPEND EPISODE_SOURCES "${GEN}/${module}.c")
  endforeach()
  add_custom_command(OUTPUT ${EPISODE_SOURCES} "${GEN}/generated.h" "${GEN}/layout.json" "${GEN}/ds.bin"
    COMMAND "${Python3_EXECUTABLE}" "${SNAPSHOT_ROOT}/tools/pas2c.py" --episode ${episode}
            --source-dir "${REPO_ROOT}/game" --placement "${SAM_PLACEMENT}" --output-dir "${GEN}"
    DEPENDS ${PASCAL_SOURCE} ${TRANSLATOR_SOURCE} "${SAM_PLACEMENT}"
    VERBATIM)
  list(APPEND GENERATED ${EPISODE_SOURCES})
endforeach()
add_executable(sam launcher.c ${GENERATED})
target_link_libraries(sam PRIVATE sam_lib sam_platform)
include(CTest)
foreach(test platform_selftest script_trace platform_demo)
  add_executable(${test} tests/${test}.c tests/test_system_bridge.c)
  target_link_libraries(${test} PRIVATE sam_platform)
endforeach()
add_executable(tp_console_integration tests/tp_console_integration.c)
target_link_libraries(tp_console_integration PRIVATE sam_platform)
add_test(NAME tp_console_integration COMMAND tp_console_integration --headless)
add_test(NAME platform_selftest COMMAND platform_selftest --headless)
add_test(NAME platform_artifacts COMMAND "${Python3_EXECUTABLE}" "${SNAPSHOT_ROOT}/tools/check_artifacts.py" "${CMAKE_BINARY_DIR}")
set_tests_properties(platform_artifacts PROPERTIES DEPENDS platform_selftest)
add_test(NAME script_determinism COMMAND "${Python3_EXECUTABLE}" "${SNAPSHOT_ROOT}/tools/check_script.py" "$<TARGET_FILE:script_trace>" "${CMAKE_CURRENT_SOURCE_DIR}/tests/trace.script")
set_tests_properties(platform_selftest tp_console_integration PROPERTIES WORKING_DIRECTORY "${CMAKE_BINARY_DIR}")
set(ERASE_GEN "${CMAKE_BINARY_DIR}/port-gen/erase")
add_custom_command(OUTPUT "${ERASE_GEN}/ERASE.c" "${ERASE_GEN}/loader.c" "${ERASE_GEN}/generated.h"
  COMMAND "${Python3_EXECUTABLE}" "${SNAPSHOT_ROOT}/tools/pas2c.py" --episode sam1
          --source-dir "${SNAPSHOT_ROOT}/tests/port/semantics" --program ERASE --output-dir "${ERASE_GEN}"
  DEPENDS "${SNAPSHOT_ROOT}/tests/port/semantics/ERASE.PAS" ${TRANSLATOR_SOURCE}
  VERBATIM)
add_executable(erase_corpus tests/corpus_runner.c "${ERASE_GEN}/ERASE.c" "${ERASE_GEN}/loader.c")
target_include_directories(erase_corpus PRIVATE "${ERASE_GEN}")
target_link_libraries(erase_corpus PRIVATE sam_platform)
